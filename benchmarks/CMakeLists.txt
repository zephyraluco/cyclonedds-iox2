cmake_minimum_required(VERSION 3.22)
project(iox2_bench LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(CycloneDDS REQUIRED)
find_package(CycloneDDS-CXX REQUIRED)
idlc_generate(TARGET bench_c_types FILES Sample.idl OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/c")
idlcxx_generate(TARGET bench_cpp_types FILES Sample.idl OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/cpp")
add_library(bench_types SHARED)
target_link_libraries(bench_types PRIVATE bench_c_types CycloneDDS::ddsc)
add_library(bench_ffi SHARED endpoint.c)
target_link_libraries(bench_ffi PRIVATE bench_types CycloneDDS::ddsc)
target_include_directories(bench_ffi PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/c")
target_include_directories(bench_ffi PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(bench_c c_main.c)
target_link_libraries(bench_c PRIVATE bench_ffi)
add_executable(bench_cpp cpp_main.cpp)
target_link_libraries(bench_cpp PRIVATE bench_cpp_types CycloneDDS-CXX::ddscxx bench_ffi)
foreach(target bench_types bench_ffi bench_c bench_cpp)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
